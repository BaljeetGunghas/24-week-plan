Challenge: Longest Consecutive Sequence

Problem Statement

Input: [100, 4, 200, 1, 3, 2]

Output: 4 (sequence: [1, 2, 3, 4])

Constraint: Must achieve O(n) time complexity

Why This Problem? It separates junior developers (brute force O(n²)) from senior developers (elegant O(n) solutions). Many still sort first (O(n log n)).

The Clever Insight

Key Discovery: For each number, only start counting a sequence if it's the BEGINNING (number - 1 doesn't exist). Otherwise, you already counted it!

Optimal O(n) Solution

function longestConsecutive(nums) {
    if (nums.length === 0) return 0;

    // Step 1: Convert to Set for O(1) lookups
    const numSet = new Set(nums);
    let longestStreak = 0;

    // Step 2: Iterate through each number
    for (const num of numSet) {
        // KEY OPTIMIZATION: Only start if it's the beginning
        if (!numSet.has(num - 1)) {
            let currentNum = num;
            let currentStreak = 1;

            // Step 3: Count consecutive sequence
            while (numSet.has(currentNum + 1)) {
                currentNum += 1;
                currentStreak += 1;
            }

            longestStreak = Math.max(longestStreak, currentStreak);
        }
    }

    return longestStreak;
}

const result = longestConsecutive([100, 4, 200, 1, 3, 2]);
console.log(result);  // Output: 4
