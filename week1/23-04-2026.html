Function 1: twoSumSorted()

Purpose

Find two indices in a sorted array that sum to target value using opposite pointers approach.

function twoSumSorted(arr, target) {
  let left = 0;
  let right = arr.length - 1;

  while (left < right) {
    let sum = arr[left] + arr[right];

    if (sum === target) {
      return [left, right];
    } else if (sum < target) {
      left++;    // Need larger number
    } else {
      right--;   // Need smaller number
    }
  }

  return null;
}

Dry Run: twoSumSorted([2, 7, 11, 15], 9)

left=0, right=3: 2+15=17 > 9 → right-- left=0, right=2: 2+11=13 > 9 → right-- left=0, right=1: 2+7=9 === 9 → Return [0, 1] ✓
Time Complexity

O(n)

Space Complexity

O(1)

Function 2-4: String & Array Operations

reverseString()

function reverseString(arr) {
  let left = 0, right = arr.length - 1;

  while (left < right) {
    [arr[left], arr[right]] = [arr[right], arr[left]];
    left++;
    right--;
  }

  return arr;
}
// Time: O(n), Space: O(1)

findMiddle() - Fast & Slow Pointers

function findMiddle(arr) {
  let slow = 0, fast = 0;

  while (fast < arr.length && arr[fast + 1] !== undefined) {
    slow++;
    fast += 2;
  }

  return arr[slow];
}
// When fast reaches end, slow is at middle

checkPalindrome()

function checkPalindrome(str) {
  let left = 0, right = str.length - 1;

  while (left < right) {
    if (str[left].toLowerCase() !== str[right].toLowerCase()) {
      return false;
    }
    left++;
    right--;
  }

  return true;
}
// Time: O(n), Space: O(1)

validPalindromeII() - Advanced

function validPalindromeII(s) {
  let left = 0, right = s.length - 1;

  while (left < right) {
    if (s[left] !== s[right]) {
      return isPalindrome(s, left + 1, right) ||
             isPalindrome(s, left, right - 1);
    }
    left++;
    right--;
  }

  return true;
}

function isPalindrome(str, left, right) {
  while (left < right) {
    if (str[left] !== str[right]) return false;
    left++;
    right--;
  }
  return true;
}

Event Loop & Async Execution

Execution Order Example

console.log("1");

setTimeout(() => console.log("2"), 0);

Promise.resolve().then(() => console.log("3"));

console.log("4");

// Output: 1, 4, 3, 2

Why This Order?

1. Call Stack (Synchronous): - console.log("1") → prints immediately - console.log("4") → prints immediately 2. Microtask Queue (High Priority): - Promise .then() → console.log("3") 3. Macrotask Queue (Lower Priority): - setTimeout → console.log("2") Event Loop Execution: 1. Execute all synchronous code 2. Check Microtask Queue 3. Check Macrotask Queue 4. Back to Microtask 5. Repeat until empty

Queue Comparison

Queue Type Examples Priority
Call Stack Synchronous Function calls Highest
Microtask Async (High) Promises, async/await High
Macrotask Async (Low) setTimeout, setInterval Low

Closures in JavaScript

function createGame() {
  let score = 0;  // Private variable

  return {
    gainPoints: () => {
      score++;
      return score;
    },
  };
}

const game = createGame();
console.log(game.gainPoints());  // 1
console.log(game.gainPoints());  // 2
// score is trapped in closure!

Common Closure Pitfall

const functions = []; for (var i = 1; i <= 3; i++) { functions.push(() => console.log(i)); } functions[0](); // Logs 4 (not 1!) functions[1](); // Logs 4 (not 2!) functions[2](); // Logs 4 (not 3!) Why: var is hoisted to function scope. All closures reference same 'i' which becomes 4 after loop! Fix: Use 'let' for block scope for (let i = 1; i <= 3; i++) { functions.push(() => console.log(i)); } // Now logs 1, 2, 3 correctly!
