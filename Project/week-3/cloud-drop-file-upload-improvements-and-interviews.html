Section 2: Security Improvements Framework

2.1 Advanced File Validation (Magic Bytes)

Current: Only checks MIME type and extension

Improved: Verify actual file signature (magic bytes)

// Check actual file content, not just extension
const fileSignatures = {
  jpg: Buffer.from([0xFF, 0xD8, 0xFF]),
  png: Buffer.from([0x89, 0x50, 0x4E, 0x47]),
  pdf: Buffer.from([0x25, 0x50, 0x44, 0x46]),
  zip: Buffer.from([0x50, 0x4B])
};

function validateFileSignature(buffer, mimeType) {
  const expectedSig = fileSignatures[getExtension(mimeType)];
  return buffer.slice(0, expectedSig.length).equals(expectedSig);
}

2.2 File Encryption (AES-256)

Implementation: Encrypt files before uploading to Cloudinary

const crypto = require('crypto');

function encryptFile(fileBuffer, encryptionKey) {
  const iv = crypto.randomBytes(16);
  const cipher = crypto.createCipheriv('aes-256-cbc', Buffer.from(encryptionKey), iv);
  let encrypted = cipher.update(fileBuffer);
  encrypted = Buffer.concat([encrypted, cipher.final()]);
  return Buffer.concat([iv, encrypted]); // Prepend IV
}

function decryptFile(encryptedBuffer, encryptionKey) {
  const iv = encryptedBuffer.slice(0, 16);
  const encrypted = encryptedBuffer.slice(16);
  const decipher = crypto.createDecipheriv('aes-256-cbc', Buffer.from(encryptionKey), iv);
  let decrypted = decipher.update(encrypted);
  return Buffer.concat([decrypted, decipher.final()]);
}

2.4 Chunked Uploads (Large Files)

Break files into 5MB chunks, upload in parallel

// Frontend - Upload in chunks
async function uploadFileInChunks(file, chunkSize = 5 * 1024 * 1024) {
  const chunks = Math.ceil(file.size / chunkSize);
  const uploadId = generateId();

  for (let i = 0; i < chunks; i++) {
    const start = i * chunkSize;
    const end = Math.min(start + chunkSize, file.size);
    const chunk = file.slice(start, end);

    const formData = new FormData();
    formData.append('uploadId', uploadId);
    formData.append('chunkIndex', i);
    formData.append('totalChunks', chunks);
    formData.append('chunk', chunk);

    await axios.post('/api/upload-chunk', formData, {
      onUploadProgress: (e) => updateProgress(i, e.loaded, e.total)
    });
  }
}

2.5 File Deduplication (SHA-256 Hash)

Detect and prevent duplicate storage

const crypto = require('crypto');

async function calculateFileHash(buffer) {
  return crypto.createHash('sha256').update(buffer).digest('hex');
}

async function checkDuplicateFile(hash, userId) {
  const existingFile = await File.findOne({
    hash: hash,
    user: userId
  });

  if (existingFile) {
    // Create hard-link instead of storing duplicate
    return { isDuplicate: true, existingFile };
  }
  return { isDuplicate: false };
}

2.6 Quota Management & Rate Limiting

// Enforce per-user quotas
const MAX_STORAGE_PER_USER = 100 * 1024 * 1024; // 100MB
const MAX_UPLOADS_PER_HOUR = 50;

async function checkUserQuota(userId) {
  const usedStorage = await File.aggregate([
    { $match: { user: userId } },
    { $group: { _id: null, total: { $sum: '$size' } } }
  ]);

  const usedBytes = usedStorage[0]?.total || 0;
  return {
    usedBytes,
    remainingBytes: MAX_STORAGE_PER_USER - usedBytes,
    quotaExceeded: usedBytes >= MAX_STORAGE_PER_USER
  };
}

2.10 Input Sanitization & Path Traversal Prevention

function sanitizeFilename(filename) {
  // Remove path traversal attempts
  filename = filename.replace(/\.\./g, '');
  filename = filename.replace(/[\/\\]/g, '');

  // Remove special characters
  filename = filename.replace(/[^a-zA-Z0-9._-]/g, '');

  // Limit length
  return filename.substring(0, 255);
}

Section 4: Complete Improved Upload Endpoint

4.1 Production-Ready Upload Endpoint

router.post("/upload", auth, rateLimiter, async (req, res) => {
  try {
    const userId = req.user._id;

    // 1. Check quota
    const quota = await checkUserQuota(userId);
    if (quota.quotaExceeded) {
      return res.status(413).json({
        message: "Storage quota exceeded",
        remainingBytes: quota.remainingBytes
      });
    }

    // 2. Validate file
    if (!req.file) return res.status(400).json({ message: "No file" });
    if (req.file.size > 1024*1024*1024) {
      return res.status(413).json({ message: "File too large" });
    }

    // 3. Validate signature (magic bytes)
    if (!validateFileSignature(req.file.buffer, req.file.mimetype)) {
      return res.status(400).json({ message: "Invalid file format" });
    }

    // 4. Scan for virus
    await scanFileForVirus(req.file.path);

    // 5. Check deduplication
    const fileHash = await calculateFileHash(req.file.buffer);
    const dup = await checkDuplicateFile(fileHash, userId);
    if (dup.isDuplicate) {
      return res.status(200).json({
        message: "File already exists",
        file: dup.existingFile
      });
    }

    // 6. Encrypt file
    const encryptedBuffer = encryptFile(req.file.buffer, process.env.ENCRYPT_KEY);

    // 7. Upload to Cloudinary
    const cloudinaryResult = await cloudinary.uploader.upload(
      encryptedBuffer,
      { resource_type: 'auto', folder: `clouddrop/${userId}` }
    );

    // 8. Save to MongoDB
    const newFile = new File({
      name: sanitizeFilename(req.file.originalname),
      type: req.file.mimetype,
      size: req.file.size,
      hash: fileHash,
      fileUrl: cloudinaryResult.secure_url,
      cloudinaryId: cloudinaryResult.public_id,
      category: getCategory(req.file.mimetype),
      user: userId,
    });

    const savedFile = await newFile.save();

    // 9. Log operation
    await logFileOperation(userId, 'file_uploaded', savedFile._id, {
      size: req.file.size,
      type: req.file.mimetype
    });

    // 10. Calculate new storage
    const maxBytes = parseInt(process.env.USER_STORAGE_LIMIT) || 104857600;
    const usedBytes = await getUsedBytes(userId);

    return res.status(201).json({
      message: "File uploaded successfully",
      statusCode: 201,
      data: {
        file: savedFile,
        storage: {
          usedBytes,
          maxBytes,
          percentage: ((usedBytes / maxBytes) * 100).toFixed(2)
        }
      }
    });

  } catch (error) {
    console.error("UPLOAD ERROR:", error);
    await logFileOperation(req.user._id, 'upload_failed', null, {
      error: error.message
    });
    return res.status(500).json({ message: error.message });
  }
});

Section 5: 50+ Interview Questions

BEGINNER LEVEL

BEGINNER Q1: What is multipart/form-data and why is it used for file uploads?

Answer: multipart/form-data is an encoding format that allows sending binary data (files) along with text fields in HTTP requests. It's used because application/x-www-form-urlencoded can't handle binary data efficiently.

Why for uploads: It preserves file structure, boundaries separate different file chunks, and it's the HTTP standard for file transmission.

BEGINNER Q2: Explain the difference between client-side and server-side file validation.

Client-side: Runs in browser, fast feedback, easily bypassed (not secure)

Server-side: Runs on backend, trustworthy, mandatory for security

Best practice: Use BOTH. Client-side for UX, server-side for security.

BEGINNER Q3: What is Multer and what problem does it solve?

Multer: Middleware for Express that handles file uploads from multipart/form-data requests.

Problems solved: Parses incoming files, saves to disk/memory, handles multiple files, enforces size limits, manages temporary storage.

INTERMEDIATE Q9: Explain how you would implement file deduplication using SHA-256.

Process:

  1. Calculate SHA-256 hash of incoming file buffer
  2. Query MongoDB for existing file with same hash + userId
  3. If found (duplicate): create hard link instead of re-uploading
  4. If not found: proceed with normal upload
  5. Store hash in File document for future lookups

Benefits: Saves 30-50% storage, reduces bandwidth, instant duplicate detection

INTERMEDIATE Q10: How would you implement chunked file uploads for large files (>1GB)?

Architecture:

  1. Frontend: Split file into 5MB chunks
  2. Generate uploadId (UUID) to track session
  3. POST each chunk with: chunkIndex, totalChunks, uploadId
  4. Backend: Store chunks to temp directory or memory
  5. On final chunk: reassemble file from chunks
  6. Verify integrity with checksum
  7. Upload complete file to Cloudinary
  8. Clean up temp storage

Benefits: 3-5x faster, resumable on network failure

Encryption in transit: Protects data while being sent over network (HTTPS/TLS)

Encryption at rest: Protects stored data on servers (AES-256 before storage)

CloudDrop should implement both: HTTPS for transmission + AES-256 before Cloudinary

Troubleshooting approach:

  • Check metrics: CPU, memory, disk I/O on backend
  • Check network: Bandwidth saturation on uplink
  • Check Cloudinary: Their API rate limits
  • Solutions: Load balancing, caching, CDN, chunking
  • Short-term: Implement rate limiting to prevent overload
  • Long-term: Auto-scaling and infrastructure upgrades
B4: Describe your process for implementing security without breaking UX.

Approach:

  • Security first: Non-negotiable (encryption, validation)
  • UX first: Make security invisible (auto-encrypt, background scan)
  • Balance: Accept slow uploads if necessary for security
  • Communicate: Show progress, not technical details
  • Example: Virus scan should happen in background, not block UI
  • Test: User testing with real files
B5: How would you design file uploads to support both small and massive files?

Strategy:

  • <100MB: Single request (fast, simple)
  • 100MB-1GB: Auto-enable chunking
  • >1GB: Mandatory chunking + resume capability
  • Frontend: Auto-detect file size, choose strategy
  • Backend: Unified endpoint handles both
  • Validation: Different limits per tier (free/pro/enterprise)
